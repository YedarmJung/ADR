\section*{기호}
\begin{itemize}
  \item 디포는 노드 $0$, 잔해 집합은 $\mathcal{N}$이다.
        아크 $a\in\mathcal{A}$는 출발 노드 $i_a$, 도착 노드 $j_a$, 출발 시각 $t^{\dep}_a$,
        도착 시각 $t^{\arr}_a$, 속도 변화 $\Delta v_a$를 가진다.
  \item 순가치 $v_i=\rho_i-\alpha_i$. $\rho_i$는 1단계에서 위험지수 $r_i$, 2단계에서 $\lambda r_i$이고,
        $\alpha_i$는 잔해 행의 쌍대가격이다.
  \item 시간 가격 $U(t)=\sum_{t_k<t}\mu_k$ ($\mu_k$: 시간 행의 쌍대가격).
        ΔV 가중치 $w$는 1단계에서 $0$, 2단계에서 $1$이다.
  \item 트립의 reduced profit은
        $\rp=\sum_{i\in S}v_i-\bigl(U(\bar e)-U(\bar s)\bigr)-w\sum\Delta v$이다.
        $\bar s=\lfloor s\rfloor_\Delta$, $\bar e=\lceil e\rceil_\Delta$는 시각 단위 $\Delta$로 내림·올림한 출발·복귀 시각이다.
  \item 추진제 조건은 $M(e^{D/c}-1)+\sum_{k\in S}p\,(e^{D_k/c}-1)\le Q$이다.
        $M$ 건조질량, $p$ 키트 질량, $Q$ 탱크 용량, $c=I_{sp}g_0$,
        $D$ 총 ΔV, $D_k$ 잔해 $k$에 도착할 때까지 쓴 ΔV.
  \item $\tau$ 잔해 작업시간, $s_0$ 디포 재장전 시간, $T$ 계획 기간,
        $K$ 트립당 최대 잔해 수, $N$ 한 번에 돌려줄 최대 열 수, $\varepsilon$ rp 허용오차.
\end{itemize}

\paragraph{라벨.}
부분 경로 하나를 $\ell=(t_\ell,\ g_\ell,\ D_\ell,\ \kappa_\ell,\ S_\ell,\ \ub_\ell,\ a_\ell,\ \mathrm{prev}_\ell)$로 요약한다.
\begin{itemize}
  \item $t_\ell$: 지금 잔해를 떠날 수 있는 가장 이른 시각 (도착 시각 $+\,\tau$)
  \item $g_\ell=\sum_{i\in S_\ell}v_i+U(\bar s)-w\,D_\ell$: 지금까지의 점수.
        복귀 아크 $a$로 닫으면 $\rp=g_\ell-U(\bar e)-w\,\Delta v_a$
  \item $D_\ell$: 지금까지의 ΔV 합
  \item $\kappa_\ell=\sum_{k\in S_\ell}p\,(e^{D_k/c}-1)$: 내려놓은 키트를 나르는 데 쓴 추진제 (키트 자체 질량은 넣지 않는다)
  \item $S_\ell$: 들른 잔해 집합, $\ub_\ell$: 이 라벨에서 나올 수 있는 rp의 상한,
        $a_\ell$: 마지막으로 탄 아크, $\mathrm{prev}_\ell$: 바로 앞 라벨
\end{itemize}

\paragraph{상한.}
$m_i=\min\{\Delta v_a: a\in\mathcal{A}',\ i_a=i\}$는 잔해 $i$에서 나갈 때 드는 최소 ΔV이다.
$E(S)$는 $S$ 밖에서 $v_i>0$인 잔해들의 $v_i$를 큰 것부터 $K-|S|$개 더한 값이다 (같은 $S$는 한 번만 계산).
잔해 $k$에 있는 라벨의 상한은 $\ub=g+E(S)-U(t)-w\,m_k$이다 ($\bar e>t$이므로 $U(\bar e)\ge U(t)$).

\paragraph{기준값 $b$.}
처음에는 $\varepsilon$이다. 모은 방문 집합이 $N$개 이상이 되면 그중 $N$번째로 큰 rp까지 올린다
($N=\infty$이면 그대로).
상한이 $b$ 이하인 라벨은 버린다.

\paragraph{시각을 지배 조건에 넣지 않는 이유.}
아크를 출발 시각 순으로 한 번만 훑는다.
라벨은 떠날 수 있는 시각 $t_\ell$이 지난 뒤에야 대기 목록 $W_j$에 들어가므로,
대기 목록의 라벨은 모두 앞으로 남은 아크를 똑같이 탈 수 있다.

\paragraph{두 모드의 차이.}
\texttt{cg.py}는 빠른 pricing을 먼저 부르고, 새 열이 없을 때만 정확한 pricing으로 확인한다.
\begin{center}
\begin{tabular}{lll}
\toprule
 & 빠른 pricing (알고리즘~\ref{alg:fast}) & 정확한 pricing (알고리즘~\ref{alg:exact}) \\
\midrule
들르는 잔해 & $v_i>0$인 잔해만 & 모든 잔해 ($v_i\le 0$인 잔해도 거쳐 간다) \\
지배의 집합 조건 & $|S_f|\le|S_\ell|$ (개수만) & $S_f\subseteq S_\ell$ \\
빈손일 때 & 개선 열이 없다는 보장이 없다 & $\rp>\varepsilon$인 트립이 정말 없다 \\
\textit{complete} & 거짓 & 참 \\
\bottomrule
\end{tabular}
\end{center}
정확한 pricing은 가장 큰 rp와 개선 열의 존재 여부를 보존한다.
모든 방문 집합의 최선 트립을 다 찾지는 않는다.

\clearpage
\begin{algbox}{빠른 pricing (\texttt{price2(\dots, use\_skip=True)})}\label{alg:fast}
\begin{algorithmic}[1]
\Require $\rho_i,\ \alpha_i\ (i\in\mathcal{N})$, 시간 가격 $U$, 아크 집합 $\mathcal{A}$, $w$, $K$, $N$, $\varepsilon$
\Ensure $\rp>\varepsilon$인 트립을 rp가 큰 순으로 최대 $N$개 (방문 집합마다 하나), \textit{complete} $=$ 거짓
\Statex
\Function{지배}{$f,\ell$} \Comment{같은 잔해에서 둘 다 떠날 수 있을 때만 부른다}
  \State \Return $g_f\ge g_\ell$ \textbf{and} $D_f\le D_\ell$ \textbf{and} $\kappa_f\le\kappa_\ell$ \textbf{and} $|S_f|\le|S_\ell|$
\EndFunction
\Statex
\Procedure{도착}{$k,t,g,D,\kappa,S,a,\ell'$} \Comment{잔해 $k$에 새 라벨이 도착한다}
  \IfThen{$M(e^{(D+m_k)/c}-1)+\kappa>Q$}{\Return} \Comment{디포로 돌아갈 추진제가 없다}
  \State $\ub\gets g+E(S)-U(t)-w\,m_k$
  \If{$\ub>b$}
    \State $\ell=(t,g,D,\kappa,S,\ub,a,\ell')$를 힙 $H$에 넣는다 \Comment{시각 $t$가 되면 $W_k$로 옮긴다}
  \EndIf
\EndProcedure
\Statex
\Procedure{빠른 pricing}{}
  \State $v_i\gets\rho_i-\alpha_i\quad(i\in\mathcal{N})$
  \State $V\gets\{i\in\mathcal{N}: v_i>0\}$ \Comment{양수 순가치 잔해만 들른다}
  \State $\mathcal{A}'\gets V\cup\{0\}$ 사이의 아크 (디포로 가는 아크는 $t^{\arr}_a+s_0\le T$인 것만), 출발 시각 순
  \State $m_i\gets\min\{\Delta v_a: a\in\mathcal{A}',\ i_a=i\}$
  \State $b\gets\varepsilon$,\quad $\mathcal{B}\gets\emptyset$,\quad $W_j\gets\emptyset\ (j\in V)$,\quad $H\gets\emptyset$ \Comment{$H$: 시각 순 힙}
  \For{$a\in\mathcal{A}'$ (출발 시각 순)}
    \While{$H$의 가장 이른 시각 $\le t^{\dep}_a$} \Comment{떠날 수 있게 된 라벨}
      \State $H$에서 잔해 $j$의 라벨 $\ell$을 꺼낸다
      \IfThen{$\ub_\ell\le b$ \textbf{or} 어떤 $f\in W_j$가 \Call{지배}{$f,\ell$}}{\textbf{continue}}
      \State $W_j$에서 \Call{지배}{$\ell,f$}인 $f$를 지우고 $\ell$을 넣는다
    \EndWhile
    \If{$i_a=0$} \Comment{디포 출발: 새 트립의 첫 라벨}
      \State $g\gets v_{j_a}+U(\lfloor t^{\dep}_a\rfloor_\Delta)-w\,\Delta v_a$
      \State \Call{도착}{$j_a,\ t^{\arr}_a+\tau,\ g,\ \Delta v_a,\ p\,(e^{\Delta v_a/c}-1),\ \{j_a\},\ a,\ \varnothing$}
    \ElsIf{$j_a=0$} \Comment{디포 복귀: 트립 완성}
      \For{$\ell\in W_{i_a}$}
        \State $D\gets D_\ell+\Delta v_a$
        \IfThen{$M(e^{D/c}-1)+\kappa_\ell>Q$}{\textbf{continue}} \Comment{탱크 초과}
        \State $\rp\gets g_\ell-U(\lceil t^{\arr}_a+s_0\rceil_\Delta)-w\,\Delta v_a$
        \If{$\rp>b$ \textbf{and} $\rp>$ ($\mathcal{B}[S_\ell]$의 rp, 없으면 $-\infty$)}
          \State $\mathcal{B}[S_\ell]\gets(\rp,a,\ell)$ \Comment{방문 집합마다 가장 좋은 것 하나}
          \IfThen{$|\mathcal{B}|\ge N$}{$b\gets\max(b,\ \mathcal{B}$에서 $N$번째로 큰 rp$)$}
        \EndIf
      \EndFor
    \Else \Comment{잔해 $i_a\to$ 잔해 $j_a$: 연장}
      \For{$\ell\in W_{i_a}$}
        \IfThen{$j_a\in S_\ell$ \textbf{or} $|S_\ell|\ge K$}{\textbf{continue}}
        \State $D\gets D_\ell+\Delta v_a$,\quad $\kappa\gets\kappa_\ell+p\,(e^{D/c}-1)$ \Comment{$j_a$에 키트를 내려놓는다}
        \State \Call{도착}{$j_a,\ t^{\arr}_a+\tau,\ g_\ell+v_{j_a}-w\,\Delta v_a,\ D,\ \kappa,\ S_\ell\cup\{j_a\},\ a,\ \ell$}
      \EndFor
    \EndIf
  \EndFor
  \State $\mathcal{B}$에서 rp가 큰 순으로 최대 $N$개를 고르고, $\mathrm{prev}$를 거꾸로 따라가며 아크를 모아 트립으로 복원한다
  \State \Return 트립, rp, \textit{complete} $=$ 거짓
\EndProcedure
\end{algorithmic}
\end{algbox}

\begin{algbox}{정확한 pricing (\texttt{price2(\dots, use\_skip=False)})}\label{alg:exact}
\begin{algorithmic}[1]
\Require $\rho_i,\ \alpha_i\ (i\in\mathcal{N})$, 시간 가격 $U$, 아크 집합 $\mathcal{A}$, $w$, $K$, $N$, $\varepsilon$
\Ensure $\rp>\varepsilon$인 트립을 rp가 큰 순으로 최대 $N$개 (방문 집합마다 하나), \textit{complete} $=$ 참
\Statex

  \State \Return $g_f\ge g_\ell$ \textbf{and} $D_f\le D_\ell$ \textbf{and} $\kappa_f\le\kappa_\ell$ \textbf{and} $S_f\subseteq S_\ell$

    \State $\ell=(t,g,D,\kappa,S,\ub,a,\ell')$를 힙 $H$에 넣는다 \Comment{시각 $t$가 되면 $W_k$로 옮긴다}
  \EndIf
\EndProcedure
\Statex
\Procedure{정확한 pricing}{}
  \State $v_i\gets\rho_i-\alpha_i\quad(i\in\mathcal{N})$
  \State $V\gets\mathcal{N}$ \Comment{모든 잔해. $v_i\le 0$인 잔해도 거쳐 갈 수 있다}

  \For{$a\in\mathcal{A}'$ (출발 시각 순)}
    \While{$H$의 가장 이른 시각 $\le t^{\dep}_a$} \Comment{떠날 수 있게 된 라벨}

    \EndWhile
    \If{$i_a=0$} \Comment{디포 출발: 새 트립의 첫 라벨}
